const std_sim = @import("std_sim");

/// A digital value the analog block reads: its path (`n`, `u.clk`), whose
/// `fieldName` is the device's `Model` field it arrives in, its slot, and
/// whether it is a §4.8 `real`.
const Name = struct { path: []const u8, slot: u32, real: bool };

/// The `digital_names` the device's card reads: Sema declares one card row
/// per digital-owned value its analog blocks read (§7.3.1), named by path.
const input_ports = blk: {
    @setEvalBranchQuota(1000 + 100 * digital_names.len);
    var n: usize = 0;
    for (digital_names) |d| {
        if (@hasField(D.Model, fieldName(d.path))) n += 1;
    }
    var out: [n]Name = undefined;
    var k: usize = 0;
    for (digital_names) |d| if (@hasField(D.Model, fieldName(d.path))) {
        out[k] = d;
        k += 1;
    };
    const final = out;
    break :blk final;
};

/// VAMS §7.3.1 Table 7-1: a bit grouping, a net and an `integer` reach the
/// continuous context as an integer, a `real` with no conversion. §7.3.2:
/// "It is an error if these operands return x or z bit values when
/// solved", and a solve is where this is read, so an x or z bit fails the
/// run by name.
fn mixedInput(dig: *std_sim.State, p: Name) f64 {
    var tmp: std.heap.ArenaAllocator = .init(std.heap.page_allocator);
    defer tmp.deinit();
    const v = std_sim.eval.read(dig, tmp.allocator(), p.slot) catch |e| mixedFail(dig, e);
    if (p.real) return @bitCast(v.values()[0]);
    const i = v.asInt() orelse {
        std.debug.print("{s}: FAIL §7.3.2: the discrete input `{s}` holds an x or z bit when the analog block reads it\n", .{ title, p.path });
        std.process.exit(1);
    };
    return @floatFromInt(i);
}

/// The potential of terminal `n1` (minus `n2`) in `x`; a name that is no
/// unknown of the device is ground.
fn potential(x: *const [n_u]f64, n1: []const u8, n2: ?[]const u8) !f64 {
    return nodeValue(x, n1) - if (n2) |b| nodeValue(x, b) else 0.0;
}
fn nodeValue(x: *const [n_u]f64, name: []const u8) f64 {
    for (u_names, 0..) |u, i| if (std.mem.eql(u8, u, name)) return x[i];
    return 0.0;
}

fn setField(model: *D.Model, comptime field: []const u8, v: anytype) void {
    @field(model, field) = std.math.lossyCast(@TypeOf(@field(model, field)), v);
}

/// The digital half stopped: its run-time error, then the run's own.
fn mixedFail(dig: *const std_sim.State, e: anyerror) noreturn {
    if (dig.failure) |f| std.debug.print("{s}\n", .{f});
    std.debug.print("FAIL {s}: the digital half did not run: {t}\n", .{ title, e });
    std.process.exit(1);
}

/// One analysis: the digital half afresh from `program_bytes`, the inputs
/// watched, and `std_sim.mixed.run` owning the global time.
fn runMixed(model: *D.Model, inst: *D.Instance, x: *[n_u]f64, forced: *[n_u]?f64, state: *State, n: *usize) void {
    var arena_state = std.heap.ArenaAllocator.init(std.heap.page_allocator);
    defer arena_state.deinit();
    const arena = arena_state.allocator();
    const p = std_sim.serial.load(arena, program_bytes) catch |e| {
        std.debug.print("FAIL {s}: the digital half does not load: {t}\n", .{ title, e });
        std.process.exit(1);
    };
    var dout = std.Io.Writer.Allocating.init(arena);
    var dig = std_sim.State.init(arena, &p, &dout.writer) catch |e| {
        std.debug.print("FAIL {s}: the digital half did not start: {t}\n", .{ title, e });
        std.process.exit(1);
    };
    var a: Analog = .{ .model = model, .inst = inst, .x = x, .forced = forced, .state = state, .dout = &dout, .n = n, .dig = &dig };
    // §8.4.3.2: an unguarded read makes the block implicitly sensitive.
    inline for (input_ports) |port| std_sim.analog.watchAnalog(&dig, port.slot) catch |e| mixedFail(&dig, e);
    // IEEE 1364 §19.8 the digital half's tick: the finest precision of all.
    std_sim.mixed.run(Analog, &a, &dig, .{ .times = &times, .tick = std_sim.time.secondsAt(1, p.finest) }) catch |e| {
        a.flush();
        mixedFail(&dig, e);
    };
    a.flush();
}

/// A `//! wave`, piecewise linear over the declared `//! time`s. At a
/// declared point it is the fixed-grid runner's value, a short list holds its
/// last entry, and an inserted point between two declared ones interpolates.
